using System;
using System.Diagnostics;
using Thermodynamics.Core;
using Thermodynamics.Harness;
using VRageMath;
class Program {
 static void Main() {
  Console.WriteLine("Open straight coolant chains: FindLoops only; min of 3 warmed runs; allocations on current thread");
  foreach(int n in new[]{128,256,512,1024,2048}) {
   var b=GridBuilder.Large(); var model=Catalog.CoolantPipeStraight();
   var orientation=PipeFitter.Orient(model,Vector3I.Left,Vector3I.Right);
   for(int i=0;i<n;i++) b.Place(model,new Vector3I(i,0,0),orientation);
   var props=LoopThermalProperties.Default();
   CoolantLoopBuilder.FindLoops(b.Grid,props,293.15f);
   double best=double.MaxValue; long allocated=0; int count=0;
   for(int r=0;r<3;r++) { long before=GC.GetAllocatedBytesForCurrentThread(); var sw=Stopwatch.StartNew();
    count=CoolantLoopBuilder.FindLoops(b.Grid,props,293.15f).Count; sw.Stop();
    long bytes=GC.GetAllocatedBytesForCurrentThread()-before; if(sw.Elapsed.TotalMilliseconds<best){best=sw.Elapsed.TotalMilliseconds;allocated=bytes;}
   }
   Console.WriteLine($"pipes={n} loops={count} ms={best:F3} allocated_bytes={allocated}");
  }
 }
}
