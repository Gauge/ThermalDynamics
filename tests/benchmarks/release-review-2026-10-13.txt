using System;
using System.Diagnostics;
using Thermodynamics.Core;
using Thermodynamics.Harness;
using Thermodynamics.Tests;
using VRageMath;
class Program {
 static void Main() {
  Console.WriteLine("scenario,pipes,diagnostics,implementation,loops,min_ms,max_ms,allocated_bytes");
  foreach(var scenario in new[]{"open","mixed","broken","closed"}) foreach(int n in new[]{128,512,2048}) {
   var b=GridBuilder.Large();
   if(scenario=="open" || scenario=="mixed") {
    var model=Catalog.CoolantPipeStraight(); var orientation=PipeFitter.Orient(model,Vector3I.Left,Vector3I.Right);
    for(int i=0;i<n;i++) b.Place(model,new Vector3I(i,0,0),scenario=="mixed" ? PipeFitter.OrientDirected(model,i%2==0?Vector3I.Right:Vector3I.Left,i%2==0?Vector3I.Left:Vector3I.Right) : orientation);
   } else {
    var ring=PipeFitter.BuildRing(b,PipeFitter.RectangleXZ(Vector3I.Zero,n/2,2));
    if(scenario=="broken") b.Grid.Remove(ring[n/2]);
   }
   foreach(bool diagnose in new[]{false,true}) foreach(bool legacy in new[]{true,false}) {
    var props=LoopThermalProperties.Default();
    Func<int> run=()=> (legacy ? LegacyCoolantLoopBuilder.FindLoops(b.Grid,props,293.15f,null,diagnose?new CoolantLoopDiagnostics():null)
      : CoolantLoopBuilder.FindLoops(b.Grid,props,293.15f,null,diagnose?new CoolantLoopDiagnostics():null)).Count;
    run(); double best=double.MaxValue,worst=0; long allocated=0; int count=0; var sw=new Stopwatch();
    for(int r=0;r<3;r++) { long before=GC.GetAllocatedBytesForCurrentThread(); sw.Restart(); count=run(); sw.Stop();
     long bytes=GC.GetAllocatedBytesForCurrentThread()-before; if(sw.Elapsed.TotalMilliseconds<best){best=sw.Elapsed.TotalMilliseconds;allocated=bytes;}
     worst=Math.Max(worst,sw.Elapsed.TotalMilliseconds);
    }
    if(count != (scenario=="closed"?1:0)) throw new Exception("Unexpected loop count");
    Console.WriteLine($"{scenario},{b.Grid.CoolantBlockCount},{diagnose},{(legacy?"legacy":"optimized")},{count},{best:F6},{worst:F6},{allocated}");
   }
  }
 }
}
